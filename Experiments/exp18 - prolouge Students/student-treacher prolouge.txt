% Rule 1: Find which teacher teaches a student
student_teacher(Student, Teacher) :-
    enrolled(Student, Code),
    teacher(Teacher, Code).

% Rule 2: Find all students of a given teacher
teacher_students(Teacher, Student) :-
    teacher(Teacher, Code),
    enrolled(Student, Code).
