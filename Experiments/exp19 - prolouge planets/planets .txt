% Rule 1: Find planets of a specific type
planets_by_type(Type, Name) :-
    planet(Name, Type, _, _).

% Rule 2: Find planets with more than a given number of moons
has_moons_greater_than(MinMoons, Name) :-
    planet(Name, _, _, Moons),
    Moons > MinMoons.
